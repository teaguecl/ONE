# one-debugger --batch smoke test on nestest (run by ctest: debugger_batch)
# nestest's automation mode starts at $C000
r pc = c000
t 3
!assert reg pc c5f9
# step over a JSR (C5FD: JSR C72D) and step back out of the subroutine
t 2
!assert reg pc c5fd
p
!assert reg pc c600

t
!assert reg pc c72d

!assert reg pc c600
# breakpoints: execute, memory write, opcode

bc *
r pc = c000

!assert reg pc c7da

# labels

# freeze: put back every frame, also after the debugger changes the byte

!assert mem 0400 77

!assert mem 0400 0
